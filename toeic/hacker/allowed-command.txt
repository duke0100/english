# Commands pre-approved for the TOEIC Hackers book builds
#
# NOTE: this file is documentation only. The list that actually stops Claude Code
# from prompting lives in  D:\workspace\english\.claude\settings.local.json
# under permissions.allow. Keep the two in sync if you add a command here.
#
# The toolkit now lives permanently in the repo:
#   D:\workspace\english\toeic\hacker\_toolkit\        (see its README.md)
# Scripts are invoked BY PATH, so the allowlist relies on the single wildcard
#   "Bash(node *)"
# rather than per-script rules. The old "Bash(node dump.js *)" style rules only
# matched a bare filename and were removed once the toolkit moved.
#
# Working files go in the session scratchpad
#   C:\Users\WINDOW~1\AppData\Local\Temp\claude\D--workspace-english\<session>\scratchpad
# Reads/writes there are allowed session-independently via
#   Read/Write/Edit(//c/Users/WINDOW~1/AppData/Local/Temp/claude/**)
# Writes into the project are limited to the TOEIC tree:
#   Write/Edit(//d/workspace/english/toeic/**)
#
# "or similar" = same tool, different page numbers / file names / crop values.

# ---------------------------------------------------------------- PDF reading
TK=/d/workspace/english/toeic/hacker/_toolkit    # path held in a variable

# Column-aware text of one or more pages of a PDF (primary way to read the book)
node $TK/dump.js "<pdf>" <pages>                    # e.g. 3   |  12-15  |  22,24
node $TK/dump.js "<pdf>" <pages> --raw --y          # same, with x/y coords (rebuild tables)
node $TK/dump.js "<pdf>" <pages> --rects            # box/rule geometry
node $TK/dump.js "<pdf>" <pages> --img              # keep image placeholders
node $TK/pdftext.js "<pdf>" QP                      # batch-extract every page to text files

# Pull a scanned page out of the PDF as a PNG so it can be viewed directly.
# Most pages of these books are scans; this is the true page image and is used
# whenever the OCR text layer is damaged or a block is missing.
node $TK/pageimg.js "<pdf>" <page> out.png
node $TK/pageimg.js "<pdf>" <page> out.png "x0,y0,x1,y1"   # crop, fractions of the page

# One-off PDF inspection helpers
node $TK/pdfinfo.js  "<pdf>"          # page count, encryption
node $TK/struct.js   "<pdf>"          # page map -> struct.txt
node $TK/keys.js     "<keypdf>"       # extract all answer keys -> keys_all.json
node $TK/pdflayout.js "<pdf>" <outdir> [from] [to]
node $TK/inv.js      "<pdf>" <from> <to>   # per-page image inventory
node $TK/bigimg.js   "<pdf>"               # list full-page scan images
node $TK/probe.js / cs.js / enc.js / dbg.js / dbg2.js "<pdf>"

# ------------------------------------------------------------------ assembly
node $TK/assemble.js <partsDir> <out.md> <keys.json>   # rebuild + verify every ✓ vs the key

# --------------------------------------------------- reading extracted text
cat QP/pNNN.txt                       # extracted page text, question book
cat KP/pNNN.txt                       # extracted page text, answer-key book
cat QP/pNNN.txt QP/pNNN.txt           # several pages at once
head -N <file>   |  tail -n +N <file>
sed -n 'A,Bp' <file>                  # print a line range
cut -c1-130 <file>
wc -l <file>     |  ls  |  ls -la

# ------------------------------------------------------- searching the text
grep -n "pattern" QP/*.txt KP/*.txt
grep -n -A<N> -B<N> "pattern" <files>
grep -o "pattern" <files> | sort | uniq -c
grep -l "pattern" <files>

# --------------------------------------- file shuffling (now wildcarded)
cp <src> <dst>                        # e.g. preserving fragments into _parts/
mv <src> <dst>
rm <file>                             # NB: Bash rules cannot be path-scoped
sed -i 's/old/new/g' <file>
mkdir -p <dir>

# --------------------------------------------------------------- git status
git status
git diff
git log --oneline -n 20

# ------------------------------------------- shell glue used around the above
# Almost every call is a compound command built from these pieces. They are what
# actually triggers a prompt when only the inner tool is allowlisted.
cd <dir>                              # working dir for every command below
Q="/d/.../QUESTION BOOK.pdf"          # path held in a variable, then "$Q"
K="/d/.../KEY BOOK.pdf"
<cmd> && <cmd>                        # chaining
<cmd> ; <cmd>
<cmd> | <cmd>                         # e.g. ... | grep / sed -n / head / tail
<cmd> 2>&1                            # merge stderr so nothing is hidden
echo "=== label ==="                  # section labels in output
node -e "…"                           # one-off inline node (JSON checks etc.)
