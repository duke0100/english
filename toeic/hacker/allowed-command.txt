# ---------------------------------------------------------------- PDF reading
# Column-aware text of one or more pages of a PDF (primary way to read the book)
node dump.js "<pdf>" <pages>                        # e.g. 3   |  12-15  |  22,24
node dump.js "<pdf>" <pages> --raw --y              # same, with x/y coordinates (rebuild tables)
node dump.js "<pdf>" <pages> --rects                # box/rule geometry
node dump.js "<pdf>" <pages> --img                  # keep image placeholders
node dump.js "<pdf>" 1-290 --out QP                 # batch-extract every page to text files

# Pull a scanned page out of the PDF as a PNG so it can be viewed directly.
# Pages 30-290 of the question book are scans; this is the true page image and
# is used whenever the OCR text layer is damaged or an option block is missing.
node pageimg.js "<pdf>" <page> out.png
node pageimg.js "<pdf>" <page> out.png "x0,y0,x1,y1"   # crop, fractions of the page

# One-off PDF inspection helpers
node pdflayout.js "<pdf>" <outdir> [from] [to]
node pdftext.js  "<pdf>" <outdir>
node keys.js     "<keypdf>"          # re-extract all 1000 answer keys
node struct.js   "<pdf>"             # page map -> struct.txt
node inv.js      "<pdf>" <from> <to> # per-page image inventory
node bigimg.js   "<pdf>"             # list full-page scan images
node probe.js / cs.js / enc.js / dbg.js / dbg2.js "<pdf>"

# ------------------------------------------------------------------ assembly
node assemble.js        # rebuild TOEIC_Hackers3_Reading.md + verify every ✓ vs the key

# --------------------------------------------------- reading extracted text
cat QP/pNNN.txt                       # extracted page text, question book
cat KP/pNNN.txt                       # extracted page text, answer-key book
cat QP/pNNN.txt QP/pNNN.txt           # several pages at once
head -N <file>   |  tail -n +N <file>

cut -c1-130 <file>
wc -l <file>     |  ls  |  ls -la

# ------------------------------------------------------- searching the text
grep -n "pattern" QP/*.txt KP/*.txt
grep -n -A<N> -B<N> "pattern" <files>
grep -o "pattern" <files> | sort | uniq -c
grep -l "pattern" <files>
